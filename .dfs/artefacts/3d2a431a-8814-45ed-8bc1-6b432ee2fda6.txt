#!/usr/bin/env python3
"""cmp.py ROWS COLS SCENARIO [driveopts...]: the same actions on claude directly in a tmux pane and through Relay;
prints, per step, whether the visible screens are equal and the diff when not."""
import subprocess, sys, time, difflib, os
T = ["/nix/store/499dwp4ljzzbx5i5fhlxm6lwzncgqz61-tmux-3.7c/bin/tmux", "-L", "w48", "-f", "/dev/null"]
CL = os.environ.get("CLAUDE_BIN", "/nix/store/3hm7i7932rk5mavvqmnzpv9x8q0nmg2a-claude-code-2.1.289/bin/claude")
def t(*a): return subprocess.run(T + list(a), capture_output=True, text=True).stdout
rows, cols, scen = int(sys.argv[1]), int(sys.argv[2]), sys.argv[3]
dopts = sys.argv[4:]
LONG = "this is a long line typed into the chat to see how the input box wraps at the edge of the terminal and again after a resize " * 2
SCEN = {
  "resize": [("wait", 2), ("resize", cols + 24, rows + 6), ("wait", 3), ("resize", cols - 20, rows - 4), ("wait", 3)],
  "type": [("wait", 2), ("type", LONG), ("wait", 3), ("resize", cols - 20, rows), ("wait", 3)],
  "none": [("wait", 2)],
}[scen]
def run(kind):
    t("kill-server"); time.sleep(1)
    if kind == "direct":
        xe = " ".join(o.split("=", 1)[1].replace(":", "=") for o in dopts if o.startswith("xenv="))
        cmd = ("cd /tmp/w48obs/home/work && env -i PATH=/usr/bin:/bin HOME=/tmp/w48obs/home TERM=xterm-256color " + xe + " "
               "ANTHROPIC_API_KEY=sk-ant-xxxxxxxxxxabcdefghijklmnopqrst %s --resume aaaaaaaa-0000-4000-8000-000000000001 "
               "--dangerously-skip-permissions; sleep 600" % CL)
        wait = 14
    else:
        cmd = "python3 /tmp/w48obs/drive.py cmp resume=1 bg=8 %s; sleep 600" % " ".join(dopts)
        wait = 14
    t("new-session", "-d", "-s", "o", "-x", str(cols), "-y", str(rows), cmd)
    time.sleep(wait)
    shots = [t("capture-pane", "-p", "-t", "o")]
    for a in SCEN:
        if a[0] == "wait": time.sleep(a[1])
        elif a[0] == "resize": t("resize-window", "-t", "o", "-x", str(a[1]), "-y", str(a[2])); time.sleep(3)
        elif a[0] == "type": t("send-keys", "-t", "o", "-l", a[1]); time.sleep(3)
        shots.append(t("capture-pane", "-p", "-t", "o"))
        info = t("display", "-p", "-t", "o", "#{pane_width}x#{pane_height} hist=#{history_size} alt=#{alternate_on}")
    return shots, info
d, di = run("direct"); r, ri = run("relay")
print("direct:", di.strip(), "| relay:", ri.strip())
for i, (a, b) in enumerate(zip(d, r)):
    step = "initial" if i == 0 else str(SCEN[i - 1])[:40]
    if a == b: print("step %d (%s): SAME" % (i, step)); continue
    print("step %d (%s): DIFFERENT" % (i, step))
    print("".join(difflib.unified_diff(a.splitlines(1), b.splitlines(1), "direct", "relay", n=1)))
if os.environ.get("SHOW"):
    for i, s in enumerate(r): print("---- relay step", i); print(s)

sys.path.insert(0, "/workspace/roadmap-dfs/scripts")

env = {"PATH": "/usr/bin:/bin", "HOME": "/tmp/w48obs/home", "TERM": "xterm-256color", "ANTHROPIC_API_KEY": "sk-ant-xxxxxxxxxxabcdefghijklmnopqrst"}

time.sleep(8)
def count(label, fn):
    n = r.total_out; fn(); time.sleep(1.5)
    d = bytes(r.tail)[n:]
    print("%-34s bytes=%5d 2J=%d H=%d" % (label, len(d), d.count(b"\x1b[2J"), d.count(b"\x1b[H")))
count("SIGWINCH, same size", lambda: os.kill(r.pid, signal.SIGWINCH))
count("resize to 24x80 (kernel SIGWINCH)", lambda: fcntl.ioctl(r.fd, termios.TIOCSWINSZ, struct.pack("HHHH", 24, 80, 0, 0)))
count("SIGWINCH again, same size", lambda: os.kill(r.pid, signal.SIGWINCH))
count("resize to 24x79", lambda: fcntl.ioctl(r.fd, termios.TIOCSWINSZ, struct.pack("HHHH", 24, 79, 0, 0)))
count("resize to 24x80", lambda: fcntl.ioctl(r.fd, termios.TIOCSWINSZ, struct.pack("HHHH", 24, 80, 0, 0)))
